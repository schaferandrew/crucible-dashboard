Let's start by creating the Ruby script `app.rb`:

```ruby
require 'json'
require 'date'

# Load the fixture cases
require_relative 'test/fixtures/cases.json'

# Load the data files
require_relative 'data/rooms.json'
require_relative 'data/rooms.csv'
require_relative 'data/guests.csv'

# Load the PHP code for the quote calculator and date helpers
require_relative 'php/quote_calculator.php'
require_relative 'php/date_helpers.php'

# Define the main app logic
def app(request)
  # Parse the request
  params = request.split('?').last.split('&').map { |param| param.split('=') }.to_h

  # Checkin date and nights
  checkin = Date.parse(params['checkin'])
  nights = params['nights'].to_i

  # Guest count
  guests = params['guests'].to_i

  # Room number
  room = params['room'].to_i

  # Calculate the quote
  quote = quote_calculator(room, checkin, nights, guests)

  # Prepare the response
  response = {
    quote: quote,
    checkin: checkin.to_s,
    nights: nights,
    guests: guests,
    room: room
  }

  # Return the response as JSON
  response.to_json
end
```

Now, let's create the `test/run_tests.rb` script to run the legacy tests:

```ruby
require 'json'
require 'date'

# Load the fixture cases
require_relative 'test/fixtures/cases.json'

# Load the PHP code for the quote calculator and date helpers
require_relative 'php/quote_calculator.php'
require_relative 'php/date_helpers.php'

# Define the main test logic
def run_tests
  cases = JSON.parse(File.read('test/fixtures/cases.json'))

  cases.each do |case|
    request = case['request']
    expected = case['expected']
    response = app(request)

    if response != expected
      puts "Test failed: request: #{request}, expected: #{expected}, got: #{response}"
      return false
    end
  end

  true
end

# Run the tests
run_tests
```
