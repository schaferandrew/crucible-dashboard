### 7. `test/run_tests.rb`

```ruby
require 'json'
require_relative '../app'

# Load fixture cases
cases = JSON.parse(File.read('test/fixtures/cases.json'))

def run_case(request)
  # Capture stdout and exit status
  out, status = Open3.capture2e('ruby', 'app.rb', request)
  [status.exitstatus, out]
end

failed = []

cases.each_with_index do |case_obj, idx|
  req   = case_obj['request']
  exp_status = case_obj['status'].to_i
  exp_body   = case_obj['body']

  act_status, act_body = run_case(req)

  unless act_status == exp_status && act_body.strip == exp_body.strip
    failed << {index: idx, request: req,
               expected_status: exp_status, actual_status: act_status,
               expected_body: exp_body.strip, actual_body: act_body.strip}
  end
end

if failed.empty?
  puts "All #{cases.size} cases passed."
  exit 0
else
  failed.each do |f|
    puts "Case #{f[:index]} FAILED:"
    puts "  Request: #{f[:request]}"
    puts "  Expected status: #{f[:expected_status]}, got: #{f[:actual_status]}"
    puts "  Expected body:\n#{f[:expected_body]}\n"
    puts "  Actual body:\n#{f[:actual_body]}\n"
  end
  exit 1
end
```
